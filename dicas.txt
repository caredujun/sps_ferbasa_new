PROCEDURE PARA RESETAR SEQUENCIA

declare
    sql varchar;
begin
    for sql in SELECT 'SELECT SETVAL(' ||
       quote_literal(quote_ident(PGT.schemaname) || '.' || quote_ident(S.relname)) ||
       ', COALESCE(MAX(' ||quote_ident(C.attname)|| '), 1) ) FROM ' ||
       quote_ident(PGT.schemaname)|| '.'||quote_ident(T.relname)|| ';'
	FROM pg_class AS S,
		 pg_depend AS D,
		 pg_class AS T,
		 pg_attribute AS C,
		 pg_tables AS PGT
	WHERE S.relkind = 'S'
		AND S.oid = D.objid
		AND D.refobjid = T.oid
		AND D.refobjid = C.attrelid
		AND D.refobjsubid = C.attnum
		AND T.relname = PGT.tablename
	ORDER BY S.relname loop

	execute sql;

	end loop;
end;

This snippet finds orphaned sequences that aren't owned by any column.

select ns.nspname as schema_name, seq.relname as seq_name
from pg_class as seq
join pg_namespace ns on (seq.relnamespace=ns.oid)
where seq.relkind = 'S'
  and not exists (select * from pg_depend where objid=seq.oid and deptype='a' and classid = 'pg_class'::regclass)
order by seq.relname;

- Exemplo de procedure no PostgreSQL retornando uma lista no procedure 'output_real_zero'.

CREATE OR REPLACE PROCEDURE public.output_real_zero(
	IN lista_in integer[],
	OUT lista integer[])
LANGUAGE 'plpgsql'
AS $BODY$
declare
	cursor1 refcursor;
BEGIN

	open cursor1 for select mae_id from fluxos_tbfluxoproducaoinputoutput as a where mae_id in (SELECT unnest(lista_in)) and (select avg(dau_valor_4) from fluxos_tbfluxoproducaoinputoutputdaugther where mae_id = a.id) = 0;
	-- O mae_id nessa tabela é o id do fluxo de produção
	Declare
		-- Vamos criar as variáveis que irão receber os dados do cursor
		v_mae_id bigint;
	begin
		loop
			FETCH NEXT FROM cursor1 into v_mae_id;
			exit when v_mae_id is null; -- quando não retornar nada sai fora do loop
			if v_mae_id not in (SELECT unnest(lista)) then
				lista = lista || v_mae_id;
			end if;
		end loop;
	end;

END;
$BODY$;

 Para ajustar sequencial de tabela

CREATE OR REPLACE FUNCTION reset_sequence(
table_name TEXT,
column_name TEXT
) RETURNS void AS $$
DECLARE
    seq_name TEXT;
    max_id INT;
BEGIN
    -- Construct the sequence name
    seq_name := table_name || '_' || column_name || '_seq';

    -- Get the maximum value of the column
    EXECUTE 'SELECT COALESCE(MAX(' || column_name || '), 1) FROM ' || table_name INTO max_id;

    -- Reset the sequence
    EXECUTE 'SELECT setval(''' || seq_name || ''', ' || max_id || ')';
END;
$$ LANGUAGE plpgsql;
